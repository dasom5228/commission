/* ========================= 오른쪽 고정 메뉴 이동 ========================= */ const menuTop = document.getElementById("menu-top"); const menuNotice = document.getElementById("menu-notice"); const menuInquiry = document.getElementById("menu-inquiry"); const mainAll = document.getElementById("main-all"); const inquirySection = document.getElementById("inquiry-section"); /* 위로 이동 */ menuTop.addEventListener("click", function() { window.scrollTo({ top: 0, behavior: "smooth" }); }); /* 안내 사항 */ menuNotice.addEventListener("click", function() { if (mainAll) { mainAll.scrollIntoView({ behavior: "smooth", block: "start" }); } }); /* 문의 양식 */ menuInquiry.addEventListener("click", function() { if (inquirySection) { inquirySection.scrollIntoView({ behavior: "smooth", block: "start" }); } });
